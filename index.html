import pygame import random import sys # Inicializa o Pygame pygame.init() # Configurações da tela WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Jogo de Pênaltis") # Cores WHITE = (255, 255, 255) GREEN = (0, 200, 0) BLACK = (0, 0, 0) # Fonte font = pygame.font.SysFont("Arial", 36) # Posições ball_pos = [WIDTH // 2, HEIGHT - 100] goalkeeper_pos = [WIDTH // 2, 150] # Variáveis ball_speed = [0, 0] shooting = False goal = None clock = pygame.time.Clock() def draw_field(): screen.fill(GREEN) pygame.draw.rect(screen, WHITE, (200, 100, 400, 50), 5) # Gol pygame.draw.circle(screen, WHITE, ball_pos, 20) # Bola pygame.draw.rect(screen, BLACK, (goalkeeper_pos[0]-40, goalkeeper_pos[1], 80, 40)) # Goleiro def reset(): global ball_pos, ball_speed, shooting, goal ball_pos = [WIDTH // 2, HEIGHT - 100] ball_speed = [0, 0] shooting = False goal = None while True: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() # Chutar com espaço if event.type == pygame.KEYDOWN: if event.key == pygame.K_SPACE and not shooting: shooting = True ball_speed = [random.randint(-5, 5), -15] # Direção aleatória goalkeeper_pos[0] = random.randint(250, 550) # Goleiro se move if shooting: ball_pos[0] += ball_speed[0] ball_pos[1] += ball_speed[1] ball_speed[1] += 1 # Gravidade # Verifica se chegou ao gol if ball_pos[1] < 150: if abs(ball_pos[0] - goalkeeper_pos[0]) < 50: goal = False # Defesa else: goal = True # Gol shooting = False draw_field() if goal is not None: text = "GOOOOL!!!" if goal else "Defendido!" msg = font.render(text, True, WHITE) screen.blit(msg, (WIDTH//2 - msg.get_width()//2, HEIGHT//2)) pygame.display.flip() pygame.time.wait(2000) reset() pygame.display.flip() clock.tick(60)
